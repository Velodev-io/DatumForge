function(spike name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE spike_common)
  if(NOT MSVC)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
