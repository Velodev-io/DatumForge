cmake_minimum_required(VERSION 3.28)
project(SolidworksCloneFeasibility VERSION 0.0.1 LANGUAGES C CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(CAD_STEP_ZERO "Build isolated Module 0 experiments only" ON)
if(NOT CAD_STEP_ZERO)
  message(FATAL_ERROR "Production modules require explicit user approval; only Module 0 exists")
endif()

find_package(Qt6 6.10.0 EXACT REQUIRED COMPONENTS Core Gui Widgets Network OpenGLWidgets Test)

  find_package(OpenCASCADE 7.9.2 EXACT REQUIRED PATHS "${CMAKE_SOURCE_DIR}/.deps/install/lib/cmake/opencascade" NO_DEFAULT_PATH)
endif()
set(DEPS "${CMAKE_SOURCE_DIR}/.deps/src")
if(NOT EXISTS "${DEPS}/sqlite/sqlite3.c" OR NOT EXISTS "${DEPS}/boost/boost/version.hpp")
  message(FATAL_ERROR "Verify/extract pinned sources with tools/build_dependencies.py first")
endif()
add_library(cad_sqlite STATIC "${DEPS}/sqlite/sqlite3.c")
target_include_directories(cad_sqlite PUBLIC "${DEPS}/sqlite")
target_compile_definitions(cad_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_FOREIGN_KEYS=1 SQLITE_OMIT_LOAD_EXTENSION=1)
add_library(cad_eigen INTERFACE)
target_include_directories(cad_eigen SYSTEM INTERFACE "${DEPS}/eigen")
include(cmake/PlaneGCS.cmake)
add_library(spike_common INTERFACE)
target_include_directories(spike_common INTERFACE "${CMAKE_SOURCE_DIR}/tests")
target_compile_definitions(spike_common INTERFACE CAD_SOURCE_ROOT="${CMAKE_SOURCE_DIR}")
function(spike name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE spike_common)
  if(NOT MSVC)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
